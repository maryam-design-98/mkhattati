/* ════════════════════════════════════════════════════ MANUAL DAY — "I'll choose today's tasks myself." A per-day override of the whole plan: the engine is set aside and only the chosen responsibilities appear. The day's energy hours are split among them by priority, once, when the choice is saved — so the times stay put while tasks are ticked off. Resets by itself tomorrow. ════════════════════════════════════════════════════ */ function manualToday(){ const m = S.manualPlan; return (m && m.date === todayISO() && Array.isArray(m.items)) ? m : null; } /* Hand out the budget 15 minutes at a time, always to whichever chosen task is furthest below its fair share by priority. Nothing gets more than it has left; micro-mode keeps its slice. */ function allocManual(ids, budget){ const items = ids.map(id => S.responsibilities.find(r => r.id === id)).filter(Boolean) .map(r => { const rem = Math.max(QUANTUM, qz(r.hours - (r.doneH || 0))); return { r, dur:0, cap: r.microMin ? Math.min(r.microMin, rem) : rem, w: PRIORITY_W[r.priority] || 1 }; }); /* everything chosen gets a first slice, even on a short day */ for(const it of items){ it.dur = Math.min(it.cap, QUANTUM); budget -= it.dur; } while(budget >= QUANTUM - 1e-9){ let best = null; for(const it of items){ if(it.dur >= it.cap - 1e-9) continue; if(!best || it.dur / it.w < best.dur / best.w) best = it; } if(!best) break; const step = Math.min(QUANTUM, best.cap - best.dur); best.dur += step; budget -= step; } return items.map(it => ({ id: it.r.id, dur: it.dur })); } /* What buildPlan() returns while a manual day is active */ function manualPlanRows(){ const cut = S.reduceTime || {}; const plan = []; for(const it of manualToday().items){ const r = S.responsibilities.find(x => x.id === it.id); if(!r) continue; /* finished & archived, or deleted */ const dur = Math.max(0, it.dur - (cut[it.id] || 0)); if(dur < 1e-6) continue; /* minus-ed away for today */ plan.push({ r, dur }); } return { plan, total: plan.reduce((s,p) => s + p.dur, 0) }; } /* ── the picker ── */ let _pickSel = new Set(); /* ids ticked in the picker */ let _pickDone = []; /* tasks already done today — always kept, with their time */ function pickBudget(){ return Math.max(0, getMaxH() - _pickDone.reduce((s,i) => s + i.dur, 0)); } function openDayPicker(){ const { plan } = buildPlan(); /* start from what's on screen now */ const done = new Set(S.completedToday); _pickDone = plan.filter(p => done.has(p.r.id)).map(p => ({ id:p.r.id, dur:p.dur })); _pickSel = new Set(plan.filter(p => !done.has(p.r.id)).map(p => p.r.id)); renderDayPicker(); document.getElementById('pick-overlay').classList.add('open'); } function closeDayPicker(){ document.getElementById('pick-overlay').classList.remove('open'); } function togglePick(id){ if(_pickSel.has(id)) _pickSel.delete(id); else _pickSel.add(id); renderDayPicker(); } function renderDayPicker(){ const done = new Set(S.completedToday); const list = pendingList().filter(r => !done.has(r.id)) .sort((a,b) => a.deadline < b.deadline ? -1 : a.deadline > b.deadline ? 1 : a.id - b.id); const preview = allocManual([..._pickSel], pickBudget()); const durOf = id => { const p = preview.find(i => i.id === id); return p ? p.dur : 0; }; const total = preview.reduce((s,i) => s + i.dur, 0) + _pickDone.reduce((s,i) => s + i.dur, 0); let html = ''; if(!list.length){ html += `
لا توجد مسؤوليات متبقية لاختيارها
`; }else{ html += `
` + list.map(r => { const c = CATS[r.cat] || CATS.other; const on = _pickSel.has(r.id); const meta = on ? fmtH(durOf(r.id)) : 'متبقٍّ ' + fmtH(r.hours - (r.doneH || 0)) + ' · حتى ' + shortArDate(r.deadline); return `
✓ ${escHtml(r.name)} ${meta}
`; }).join('') + `
`; } html += `
${_pickSel.size ? 'المختار: ' + toAr(_pickSel.size) + ' · ' + fmtH(total) + ' من ' + fmtH(getMaxH()) : 'اختاري مهمة واحدة على الأقل'}
${manualToday() ? `` : ''}
`; document.getElementById('pick-body').innerHTML = html; } function saveDayPicker(){ if(!_pickSel.size) return; S.manualPlan = { date: todayISO(), items: allocManual([..._pickSel], pickBudget()).concat(_pickDone) }; S.reduceTime = {}; /* fresh times — old "−" cuts no longer apply */ saveS(); closeDayPicker(); renderPlanLines(); } function resetDayToAuto(){ S.manualPlan = null; S.reduceTime = {}; saveS(); closeDayPicker(); renderPlanLines(); }
